#include <stdio.h>
#include <stdlib.h>
#include <math.h>
void main()
{ // программа считает

  system("chcp 1251");
  printf("Введите границы интервала и шаг через пробел \n");
  scanf("%lf%lf%lf", &a, &b, &h);
  x = a;
  while (x <= b)

  }
